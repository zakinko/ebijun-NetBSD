# Checks that every NetBSD image in this repository must pass.
#
# Concatenated ahead of the per-port file by the workflow, so keep anything
# port-specific (device names, architecture, boot partition) out of here.

# --- the system is actually NetBSD and reached multi-user -----------------
expect ^NetBSD              :: uname -s
run                            test -c /dev/null
expect ^/                   :: sysctl -n kern.rootdev 2>/dev/null || mount | sed -n 's/ on \/ .*//p'

# rc(8) writes this only once it has run the whole /etc/rc.d set.  Its
# absence is the signature of a boot that stopped at single user.
run                            test -s /var/run/dmesg.boot

# --- nothing in the kernel log that should not be there -------------------
absent panic                :: dmesg | grep -i panic
absent 'WARNING: .* preposterous' :: dmesg
absent 'uvm_fault'          :: dmesg

# --- the root filesystem came up read-write -------------------------------
absent 'read-only'          :: mount | grep ' on / '
run                            touch /.ci-write-probe
run                            rm -f /.ci-write-probe

# --- the filesystem is not damaged ----------------------------------------
# fsck -n on the live root reports rather than repairs; any "UNREF" or
# "UNALLOCATED" here means the image was built from a filesystem that was
# not cleanly unmounted.
absent 'UNREF|UNALLOCATED|BAD SUPER BLOCK' :: fsck -n / 2>&1 || true

# --- users and permissions ------------------------------------------------
expect ^root:               :: grep '^root:' /etc/master.passwd
expect 0                    :: id -u

# --- time is sane, so pkgin and certificate checks will work --------------
expect ^20                  :: date +%Y
